# Pinned toolchain for regenerating requirements-build.txt (hermetic builds).
#
# requirements-build.txt lists PEP-517 build-time dependencies and is produced by:
#
#     python -m venv toolenv
#     . toolenv/bin/activate
#     pip install -r requirements-tooling.txt
#     pybuild-deps compile --generate-hashes --output-file=requirements-build.txt
#
# Why these exact pins:
#   - pybuild-deps 0.5.0 (latest) passes `generate_hashes` to pip-tools'
#     OutputWriter.__init__, which pip-tools >=7.6.0 removed -> TypeError.
#   - pip-tools 7.5.0 still accepts that argument, but its resolver calls
#     pip's make_requirement_preparer() without `allow_editables`, which pip
#     >=25.1 made a required keyword -> TypeError.
#   - So pip must stay <25.1 to work with pip-tools 7.5.0.
# This trio is the only mutually compatible combination until pybuild-deps
# ships a release compatible with pip-tools >=7.6.
#
# If the pybuild-deps cache ever gets corrupted ("dbm.error: db type could
# not be determined"), clear it: rm -rf ~/.cache/pybuild-deps

pip<25.1
pip-tools==7.5.0
pybuild-deps==0.5.0
